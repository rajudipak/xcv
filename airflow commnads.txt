import pendulum
from datetime import datetime
from zoneinfo import ZoneInfo
from airflow.sdk import dag, task

@dag(
    dag_id="date_time",
    schedule="30 9 * * 3",
    start_date=pendulum.datetime(2026, 10, 26, tz="Asia/Kolkata"),
    catchup=False
)
def date_time():

    @task
    def show_date_time():
        print(datetime.now(ZoneInfo("Asia/Kolkata")))

    show_date_time()

date_time()
