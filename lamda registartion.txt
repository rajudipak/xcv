
import json
import boto3
import uuid

table = boto3.resource('dynamodb').Table('Register')

def lambda_handler(event, context):
    data = json.loads(event['body'])

    table.put_item(Item={
        'id': str(uuid.uuid4()),
        'name': data['name'],
        'rollno': data['rollno'],
        'branch': data['branch']
    })

    return {
        'statusCode': 200,
        'headers': {'Content-Type': 'application/json'},
        'body': json.dumps({
            'message': 'Registration successful'
        })
    }
